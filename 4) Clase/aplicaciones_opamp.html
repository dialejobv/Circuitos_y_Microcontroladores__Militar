Juegos en una OLED de 128 × 64 con comparadores

El joystick (potenciómetro) pasa por un seguidor; dos comparadores con umbrales de 1,67 V y 3,33 V entregan izquierda y derecha al microcontrolador. Use las flechas ← → o el deslizador.

Esquema del circuito

Pantalla OLED

Joystick y comparadores

C2 izquierda (v < 1,67 V) C1 derecha (v > 3,33 V) ADC (Pong, 10 bits)

Histéresis de ±25 mV (RF = 1 MΩ) alrededor de cada umbral: el jugador no ve temblores.

Sketch de Arduino: Esquiva (Adafruit SSD1306 + GFX)

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

Adafruit_SSD1306 oled(128, 64, &Wire, -1);
const int PIN_DER = 2;   // salida de C1 (LM393 + pull-up 10 kΩ): alto si v > 3,33 V
const int PIN_IZQ = 3;   // salida de C2: alto si v < 1,67 V

struct Bloque { int x, y, w; };
Bloque b[5];
int x = 60;
long puntos = 0;
bool fin = false;

void nuevo(Bloque &q) { q.w = random(6, 16); q.x = random(0, 128 - q.w); q.y = random(-60, 0); }

void reiniciar() { x = 60; puntos = 0; fin = false; for (auto &q : b) nuevo(q); }

void setup() {
  pinMode(PIN_DER, INPUT);
  pinMode(PIN_IZQ, INPUT);
  oled.begin(SSD1306_SWITCHCAPVCC, 0x3C);
  oled.setTextColor(SSD1306_WHITE);
  randomSeed(analogRead(A1));
  reiniciar();
}

void loop() {
  bool der = digitalRead(PIN_DER), izq = digitalRead(PIN_IZQ);
  if (!fin) {
    x += der ? 3 : (izq ? -3 : 0);
    x = constrain(x, 0, 120);
    int vel = 1 + puntos / 150;            // cada vez más rápido
    for (auto &q : b) {
      q.y += vel;
      if (q.y > 63) { nuevo(q); puntos += 10; }
      if (q.y + 6 >= 54 && q.x < x + 8 && q.x + q.w > x) fin = true;   // choque
    }
  } else if (izq && der == false) {
    delay(300); reiniciar();               // joystick a la izquierda: jugar otra vez
  }
  oled.clearDisplay();
  oled.setCursor(0, 0); oled.print("PUNTOS "); oled.print(puntos);
  oled.drawFastHLine(0, 9, 128, SSD1306_WHITE);
  for (auto &q : b) if (q.y > 10) oled.fillRect(q.x, q.y, q.w, 6, SSD1306_WHITE);
  oled.fillTriangle(x, 63, x + 4, 54, x + 8, 63, SSD1306_WHITE);
  if (fin) { oled.setCursor(46, 30); oled.print("FIN"); }
  oled.display();
  delay(30);
}
